cmake_minimum_required(VERSION 3.25)
if(APPLE)
  set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "Mac architectures")
endif()
project(SeparateSong VERSION 1.1.0 LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(${PROJECT_NAME} SHARED src/main.cpp src/Bridge.cpp src/JukeboxLink.cpp src/AudioTap.cpp src/EffectsPlayback.cpp src/OffsetSetting.cpp src/PauseVolume.cpp src/LevelIdentity.cpp)
if(APPLE)
  enable_language(OBJCXX)
  target_sources(${PROJECT_NAME} PRIVATE src/MacInput.mm)
  target_link_libraries(${PROJECT_NAME} "-framework Cocoa")
endif()
target_include_directories(${PROJECT_NAME} PRIVATE vendor/jukebox-3.8.0)
if(WIN32)
  target_compile_definitions(${PROJECT_NAME} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(${PROJECT_NAME} PRIVATE /utf-8 /bigobj)
  target_link_libraries(${PROJECT_NAME} ws2_32)
endif()
if(NOT DEFINED ENV{GEODE_SDK})
  message(FATAL_ERROR "Set GEODE_SDK to the Geode SDK directory")
endif()
add_subdirectory($ENV{GEODE_SDK} ${CMAKE_BINARY_DIR}/geode)
list(APPEND GEODE_MODS_BEING_BUILT fleym.nongd)
setup_geode_mod(${PROJECT_NAME} DONT_INSTALL)
