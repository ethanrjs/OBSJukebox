cmake_minimum_required(VERSION 3.25)
project(SeparateSong LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(OBS_JUKEBOX_QA "Enable test-only environment overrides" OFF)
add_library(${PROJECT_NAME} SHARED src/main.cpp src/Bridge.cpp src/JukeboxLink.cpp src/AudioTap.cpp src/EffectsPlayback.cpp src/OffsetSetting.cpp src/PauseVolume.cpp src/LevelIdentity.cpp src/Shutdown.cpp)
if(APPLE)
  enable_language(OBJCXX)
  target_sources(${PROJECT_NAME} PRIVATE src/MacInput.mm)
  target_link_libraries(${PROJECT_NAME} "-framework Cocoa")
endif()
# The Jukebox version whose vendored headers the mod is built against. JukeboxLink turns the
# integration off on any other version, since it reads Jukebox's fields at these headers' offsets.
set(JUKEBOX_VERSION 3.8.0)
target_include_directories(${PROJECT_NAME} PRIVATE vendor/jukebox-${JUKEBOX_VERSION})
target_compile_definitions(${PROJECT_NAME} PRIVATE JUKEBOX_VERSION="${JUKEBOX_VERSION}")
if(OBS_JUKEBOX_QA)
  target_compile_definitions(${PROJECT_NAME} PRIVATE OBS_JUKEBOX_QA=1)
endif()
if(MSVC)
  target_compile_options(${PROJECT_NAME} PRIVATE /W4)
else()
  target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra)
endif()
if(WIN32)
  target_compile_definitions(${PROJECT_NAME} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(${PROJECT_NAME} PRIVATE /utf-8 /bigobj)
  target_link_libraries(${PROJECT_NAME} ws2_32)
endif()
if(NOT DEFINED ENV{GEODE_SDK})
  message(FATAL_ERROR "Set GEODE_SDK to the Geode SDK directory")
endif()
add_subdirectory($ENV{GEODE_SDK} ${CMAKE_BINARY_DIR}/geode)
# Jukebox comes from the vendored headers rather than the Geode index. Node IDs is an optional
# runtime dependency, used only for its node IDs, so the build doesn't fetch it either.
setup_geode_mod(${PROJECT_NAME} DONT_INSTALL EXTERNALS fleym.nongd:v${JUKEBOX_VERSION} geode.node-ids:v1.23.0)
