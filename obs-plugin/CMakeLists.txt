cmake_minimum_required(VERSION 3.25)
project(SeparateSongOBS LANGUAGES CXX C)
if(APPLE)
  message(FATAL_ERROR "Use scripts/build-obs-plugin.sh for the macOS plugin")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(separate-song MODULE separate-song.cpp Decoder.cpp)
# The plugin shows its release version in the source properties. mod.json is the release's single
# source of truth, so read the version from it here rather than duplicating the number in the code.
set(OBS_JUKEBOX_VERSION "dev")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../mod.json")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../mod.json" _obs_jukebox_mod_json)
  string(REGEX MATCH "\"version\"[ \t]*:[ \t]*\"([^\"]+)\"" _obs_jukebox_version_match "${_obs_jukebox_mod_json}")
  if(CMAKE_MATCH_1)
    set(OBS_JUKEBOX_VERSION "${CMAKE_MATCH_1}")
  endif()
endif()
target_compile_definitions(separate-song PRIVATE OBS_JUKEBOX_VERSION="${OBS_JUKEBOX_VERSION}")
option(OBS_JUKEBOX_QA "Enable isolated test ports and diagnostic capture hooks" OFF)
if(OBS_JUKEBOX_QA)
  target_compile_definitions(separate-song PRIVATE OBS_JUKEBOX_QA)
endif()
if(WIN32)
  set(OBS_SDK "${CMAKE_CURRENT_SOURCE_DIR}/../tools/obs-sdk" CACHE PATH "OBS source headers")
  set(SIMDE_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../tools/simde" CACHE PATH "SIMDe headers")
  set(OBS_IMPORT_LIBRARY "" CACHE FILEPATH "Import library generated from the installed OBS DLL")
  if(NOT EXISTS "${OBS_IMPORT_LIBRARY}")
    message(FATAL_ERROR "Set OBS_IMPORT_LIBRARY to the import library for your OBS version")
  endif()
  target_include_directories(separate-song PRIVATE "${OBS_SDK}/libobs" "${SIMDE_INCLUDE}" "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_definitions(separate-song PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
  target_compile_options(separate-song PRIVATE /bigobj /utf-8 /W4)
  target_link_libraries(separate-song PRIVATE "${OBS_IMPORT_LIBRARY}" ws2_32 mfplat mfreadwrite mfuuid ole32)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(LIBOBS REQUIRED IMPORTED_TARGET libobs)
  find_package(Threads REQUIRED)
  target_link_libraries(separate-song PRIVATE PkgConfig::LIBOBS Threads::Threads ${CMAKE_DL_LIBS} m)
  target_compile_options(separate-song PRIVATE $<$<CONFIG:Release>:-O2> -Wall -Wextra)
  install(TARGETS separate-song LIBRARY DESTINATION separate-song/bin/64bit)
else()
  message(FATAL_ERROR "Unsupported platform")
endif()
set_target_properties(separate-song PROPERTIES PREFIX "")
