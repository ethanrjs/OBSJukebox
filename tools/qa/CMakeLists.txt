cmake_minimum_required(VERSION 3.25)
project(SeparateSongRuntimeQA LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(game-identity-validation game-identity-validation.cpp)
add_test(NAME game-identity-validation COMMAND game-identity-validation)
set(OBS_SDK "${CMAKE_CURRENT_SOURCE_DIR}/../obs-sdk" CACHE PATH "OBS source headers")
set(SIMDE_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../simde" CACHE PATH "SIMDe headers")
add_executable(obs-runtime-validation obs-runtime-validation.cpp)
target_include_directories(obs-runtime-validation PRIVATE ../../obs-plugin)
find_package(Threads REQUIRED)
target_link_libraries(obs-runtime-validation PRIVATE Threads::Threads)
add_executable(decoder-seek-validation decoder-seek-validation.cpp ../../obs-plugin/Decoder.cpp)
add_executable(decoder-format-validation decoder-format-validation.cpp ../../obs-plugin/Decoder.cpp)
foreach(decoder_target IN ITEMS decoder-seek-validation decoder-format-validation)
  target_link_libraries(${decoder_target} PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
endforeach()
if(WIN32)
  set(OBS_IMPORT_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../../build-windows/obs.lib" CACHE FILEPATH "OBS import library")
  target_include_directories(obs-runtime-validation PRIVATE "${OBS_SDK}/libobs" "${SIMDE_INCLUDE}")
  target_compile_definitions(obs-runtime-validation PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(obs-runtime-validation PRIVATE /EHsc /utf-8)
  target_link_libraries(obs-runtime-validation PRIVATE "${OBS_IMPORT_LIBRARY}" ws2_32)
  target_compile_definitions(decoder-seek-validation PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_compile_options(decoder-seek-validation PRIVATE /EHsc /utf-8)
  target_link_libraries(decoder-seek-validation PRIVATE mfplat mfreadwrite mfuuid ole32)
  target_compile_definitions(decoder-format-validation PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_compile_options(decoder-format-validation PRIVATE /EHsc /utf-8)
  target_link_libraries(decoder-format-validation PRIVATE mfplat mfreadwrite mfuuid ole32)
  add_executable(fmod-tap-validation fmod-tap-validation.cpp)
  target_compile_definitions(fmod-tap-validation PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
elseif(APPLE)
  set(OBS_APP "/Applications/OBS.app" CACHE PATH "OBS application")
  find_library(OBS_LIBRARY NAMES libobs PATHS "${OBS_APP}/Contents/Frameworks" NO_DEFAULT_PATH REQUIRED)
  target_include_directories(obs-runtime-validation PRIVATE "${OBS_SDK}/libobs" "${SIMDE_INCLUDE}")
  target_link_libraries(obs-runtime-validation PRIVATE "${OBS_LIBRARY}")
  target_compile_definitions(obs-runtime-validation PRIVATE OBS_QA_DATA_PATH="${OBS_APP}/Contents/Frameworks/libobs.framework/Versions/A/Resources/" OBS_QA_GRAPHICS_MODULE="${OBS_APP}/Contents/Frameworks/libobs-opengl.dylib")
  set_target_properties(obs-runtime-validation PROPERTIES BUILD_RPATH "${OBS_APP}/Contents/Frameworks")
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(LIBOBS REQUIRED IMPORTED_TARGET libobs)
  target_link_libraries(obs-runtime-validation PRIVATE PkgConfig::LIBOBS)
endif()

find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(protocol-layout protocol-layout.cpp)
add_test(NAME protocol-layout-parity COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_protocol_layout.py $<TARGET_FILE:protocol-layout>)
foreach(qa_target IN ITEMS game-identity-validation obs-runtime-validation decoder-seek-validation decoder-format-validation protocol-layout)
  target_compile_definitions(${qa_target} PRIVATE OBS_JUKEBOX_QA=1)
  if(MSVC)
    target_compile_options(${qa_target} PRIVATE /W4)
  else()
    target_compile_options(${qa_target} PRIVATE -Wall -Wextra)
  endif()
endforeach()

foreach(mode IN ITEMS release qa)
  add_executable(transport-${mode} ../../scripts/tests/transport/tests.cpp)
  if(mode STREQUAL "qa")
    target_compile_definitions(transport-${mode} PRIVATE OBS_JUKEBOX_QA=1)
  endif()
  if(WIN32)
    target_link_libraries(transport-${mode} PRIVATE ws2_32)
  endif()
  add_test(NAME transport-${mode} COMMAND transport-${mode})
endforeach()
